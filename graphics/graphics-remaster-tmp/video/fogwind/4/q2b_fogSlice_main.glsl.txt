// Agent 4 (fogwind) variant q2b: each cell integrates its whole depth span with
// 2 fixed taps (metric weights), tap i at its own depth stratum and its own fixed point
// of the cell; no temporal jitter. The history is a smoother only.
const int KT = 2;
const vec2 kTapXY[KT] = vec2[KT](vec2(-0.250000, -0.250000), vec2(0.250000, 0.250000));
const float kTapZ[KT] = float[KT](0.250000, 0.750000);
vec3 g_tapDir[KT];
vec4 fogSlice(float s, vec3 dirZ) {
    vec4 acc = vec4(0.0);
    float wsum = 0.0;
    const float h = 0.5 / float(KT);
    for (int i = 0; i < KT; ++i) {
        float w = fogSliceDepth((s + kTapZ[i] + h) / fogGrid.z) - fogSliceDepth(max(s + kTapZ[i] - h, 0.0) / fogGrid.z);
        float z = fogSliceDepth(max(s + kTapZ[i], 0.0) / fogGrid.z);
        acc += w * fogCell(g_tapDir[i] * z);
        wsum += w;
    }
    vec4 cur = acc / max(wsum, 1e-6);
    vec3 rel = dirZ * fogSliceDepth((s + 0.5) / fogGrid.z);
    float share = fogFrame.z;
    if (share < 1.0) {
        vec3 pv = vec3(dot(prevRow0.xyz, rel) + prevRow0.w, dot(prevRow1.xyz, rel) + prevRow1.w,
                       dot(prevRow2.xyz, rel) + prevRow2.w);
        if (pv.z > 0.05) {
            vec2 puv = pv.xy * prevProj.xy / pv.z * 0.5 + 0.5;
            float pt = prevProj.w < 1e-3 ? pv.z / prevProj.z
                                         : log(1.0 + pv.z * (exp(prevProj.w) - 1.0) / prevProj.z) / prevProj.w;
            if (puv.x >= 0.0 && puv.x <= 1.0 && puv.y >= 0.0 && puv.y <= 1.0 && pt <= 1.0)
                cur = mix(0.25 * (texture(fogHistory, vec3(puv, pt) + vec3(0.25) / fogGrid.xyz * vec3(1.0, 1.0, 1.0)) + texture(fogHistory, vec3(puv, pt) + vec3(0.25) / fogGrid.xyz * vec3(1.0, -1.0, -1.0)) + texture(fogHistory, vec3(puv, pt) + vec3(0.25) / fogGrid.xyz * vec3(-1.0, 1.0, -1.0)) + texture(fogHistory, vec3(puv, pt) + vec3(0.25) / fogGrid.xyz * vec3(-1.0, -1.0, 1.0))), cur, share);
        }
    }
    return cur;
}

    }
    vec2 ndc = gl_FragCoord.xy / fogGrid.xy * 2.0 - 1.0;
