float Fract(float x)
{
    return x - std::floor(x);
}

// puddleHash and puddleNoise of the puddles piece.
float PuddleHashMirror(float px, float py)
{
    float x = Fract(px * 0.1031f), y = Fract(py * 0.1031f), z = Fract(px * 0.1031f);
    const float d = x * (y + 33.33f) + y * (z + 33.33f) + z * (x + 33.33f);
    x += d;
    y += d;
    z += d;
    return Fract((x + y) * z);
}

float PuddleNoiseMirror(float px, float py)
{
    const float ix = std::floor(px), iy = std::floor(py);
    const float fx = px - ix, fy = py - iy;
    const float ux = fx * fx * (3.0f - 2.0f * fx), uy = fy * fy * (3.0f - 2.0f * fy);
    const float a = PuddleHashMirror(ix, iy), b = PuddleHashMirror(ix + 1.0f, iy);
    const float c = PuddleHashMirror(ix, iy + 1.0f), d = PuddleHashMirror(ix + 1.0f, iy + 1.0f);
    const float lo = a + (b - a) * ux;
    const float hi = c + (d - c) * ux;
    return lo + (hi - lo) * uy;
}

// GLSL mat2(a, b, c, d) * v (columns (a, b) and (c, d)), scaled and offset.
float NoiseTurned(float a, float b, float c, float d, float vx, float vy, float scale, float offset)
{
    return PuddleNoiseMirror((a * vx + c * vy) / scale + offset, (b * vx + d * vy) / scale + offset);
}

// puddleRelief with every octave in (a pixel far finer than the finest).
float PuddleReliefMirror(float x, float z)
{
    const float px = x + (PuddleNoiseMirror(x / 5.3f + 7.1f, z / 5.3f + 7.1f) - 0.5f) * 2.4f;
    const float pz = z + (PuddleNoiseMirror(x / 5.3f + 23.9f, z / 5.3f + 23.9f) - 0.5f) * 2.4f;
    return NoiseTurned(0.8f, 0.6f, -0.6f, 0.8f, px, pz, 3.5f, 0.0f) * 0.55f +
           NoiseTurned(0.28f, -0.96f, 0.96f, 0.28f, px, pz, 1.3f, 17.0f) * 0.3f +
           NoiseTurned(0.6f, 0.8f, -0.8f, 0.6f, px, pz, 0.45f, 41.0f) * 0.15f;
}

// B7's first puddle noise, kept for comparison: the same octaves on the
// world's axes, without the warp.
float PuddleAxisNoise(float x, float z)
{
    return PuddleNoiseMirror(x / 3.5f, z / 3.5f) * 0.55f + PuddleNoiseMirror(x / 1.3f + 17.0f, z / 1.3f + 17.0f) * 0.3f +
           PuddleNoiseMirror(x / 0.45f + 41.0f, z / 0.45f + 41.0f) * 0.15f;
}

// A height grid as the shader reads it (puddleHeight): the texels between
// smoothstep weights, clamped at the map's edges.
struct HeightGrid
{
    int n = 0;
    float cell = 12.5f;
    std::vector<float> h;
    float At(int x, int z) const
    {
        x = std::clamp(x, 0, n - 1);
        z = std::clamp(z, 0, n - 1);
        return h[static_cast<size_t>(z) * n + x];
    }
    float Height(float wx, float wz) const
    {
        const float tx = wx / cell, tz = wz / cell;
        const float ix = std::floor(tx), iz = std::floor(tz);
        float fx = tx - ix, fz = tz - iz;
        fx = fx * fx * (3.0f - 2.0f * fx);
        fz = fz * fz * (3.0f - 2.0f * fz);
        const int x0 = static_cast<int>(ix), z0 = static_cast<int>(iz);
        const float a = At(x0, z0) + (At(x0 + 1, z0) - At(x0, z0)) * fx;
        const float b = At(x0, z0 + 1) + (At(x0 + 1, z0 + 1) - At(x0, z0 + 1)) * fx;
        return a + (b - a) * fz;
    }
    // puddleBasin.
    float Basin(float x, float z, float& slope) const
    {
        const float h0 = Height(x, z);
        const float hx0 = Height(x - cell, z), hx1 = Height(x + cell, z);
        const float hz0 = Height(x, z - cell), hz1 = Height(x, z + cell);
        slope = std::sqrt((hx1 - hx0) * (hx1 - hx0) + (hz1 - hz0) * (hz1 - hz0)) * (0.5f / cell);
        const float c = 2.5f * cell;
        const float wide = Height(x + c, z + c) + Height(x - c, z + c) + Height(x + c, z - c) + Height(x - c, z - c);
        return (hx0 + hx1 + hz0 + hz1) * 0.25f - h0 + (wide * 0.25f - h0) * 0.25f;
    }
};

template <typename F> HeightGrid MakeGrid(F height)
{
    HeightGrid g;
    g.n = 48;
    g.h.resize(static_cast<size_t>(g.n) * g.n);
    for (int z = 0; z < g.n; ++z)
        for (int x = 0; x < g.n; ++x)
            g.h[static_cast<size_t>(z) * g.n + x] = height(x * g.cell, z * g.cell);
    return g;
}

struct PuddleConstants
{
    float ground, edge, basin, basinDepth, slope, slopeFree, slopeStart, slopeEnd;
};

PuddleConstants ReadPuddleConstants(const std::string& puddles)
{
    const auto constant = [&](const char* name)
    {
        const std::regex re(std::string("const float ") + name + R"( = ([0-9.]+);)");
        std::smatch m;
        INFO(name);
        REQUIRE(std::regex_search(puddles, m, re));
        return std::stof(m[1].str());
    };
    return {constant("kPuddleGround"),     constant("kPuddleEdge"),       constant("kPuddleBasin"),
            constant("kPuddleBasinDepth"), constant("kPuddleSlope"),      constant("kPuddleSlopeFree"),
            constant("kPuddleSlopeStart"), constant("kPuddleSlopeEnd")};
}

float Smooth(float e0, float e1, float v)
{
    const float t = std::clamp((v - e0) / (e1 - e0), 0.0f, 1.0f);
    return t * t * (3.0f - 2.0f * t);
}

// wetPuddle's standing water on bare ground at the full puddle level, near
// the eye.
float PuddleWaterMirror(const HeightGrid& g, const PuddleConstants& k, float x, float z)
{
    float slope = 0.0f;
    const float field =
        PuddleReliefMirror(x, z) + k.basin * std::clamp(g.Basin(x, z, slope) / k.basinDepth, -1.0f, 1.0f);
    const float th = k.ground + k.slope * std::max(slope - k.slopeFree, 0.0f);
    return Smooth(th, th + k.edge, field) * (1.0f - Smooth(k.slopeStart, k.slopeEnd, slope));
}

// The share of standing water over a 60 m patch in the middle of the grid.
float WaterShare(const HeightGrid& g, const PuddleConstants& k)
{
    const float c = 0.5f * g.n * g.cell;
    double sum = 0.0;
    int count = 0;
    for (float z = c - 30.0f; z < c + 30.0f; z += 0.25f)
        for (float x = c - 30.0f; x < c + 30.0f; x += 0.25f, ++count)
            sum += PuddleWaterMirror(g, k, x, z);
    return static_cast<float>(sum / count);
}

// Of the water lines a field draws at `th`, the share running within 5 degrees
// of the world's axes (over an even spread of directions: 1/9).
template <typename F> float AxisAlignedShare(F field, float th)
{
    int lines = 0, aligned = 0;
    const float step = 0.1f;
    for (float z = 1000.0f; z < 1080.0f; z += step)
        for (float x = 2000.0f; x < 2080.0f; x += step)
        {
            const float v = field(x, z);
            const float vx = field(x + step, z), vz = field(x, z + step);
            if ((v - th) * (vx - th) >= 0.0f && (v - th) * (vz - th) >= 0.0f)
                continue;
            const float gx = vx - v, gz = vz - v;
            const float deg = std::atan2(std::fabs(gz), std::fabs(gx)) * 57.29578f; // 0..90
            ++lines;
            if (deg < 5.0f || deg > 85.0f)
                ++aligned;
        }
    REQUIRE(lines > 1000);
    return static_cast<float>(aligned) / lines;
}

TEST_CASE("Wet weather puddles: water stands where the ground's shape collects it", "[remaster][wet][TerrainHorizon]")
{
    const std::string puddles = gl::RemasterFragmentPiece(gl::FragmentPiece::Puddles);
    const PuddleConstants k = ReadPuddleConstants(puddles);
    // The shape enters the field: the basin raises it, the slope raises the
    // threshold and fades the water out.
    CHECK(puddles.find("float puddleBasin(vec2 xz, out float slope) {") != std::string::npos);
    CHECK(puddles.find("float field = puddleRelief(posW.xz, footT);") != std::string::npos);
    CHECK(puddles.find("field += kPuddleBasin * clamp(puddleBasin(posW.xz, slope) / kPuddleBasinDepth, -1.0, 1.0);") !=
          std::string::npos);
    CHECK(puddles.find("kPuddleSlope * max(slope - kPuddleSlopeFree, 0.0);") != std::string::npos);
    CHECK(puddles.find("(1.0 - smoothstep(kPuddleSlopeStart, kPuddleSlopeEnd, slope))") != std::string::npos);
    CHECK(k.slopeStart < k.slopeEnd);

    // A bowl, level ground, a crest and planes of rising slope, on the 12.5 m
    // grid of a remastered island. The bowl and the crest are 0.3 m deep over
    // the 60 m patch.
    const float mid = 24.0f * 12.5f;
    const auto r2 = [mid](float x, float z) { return (x - mid) * (x - mid) + (z - mid) * (z - mid); };
    const HeightGrid bowl = MakeGrid([&](float x, float z) { return 0.3f * r2(x, z) / 900.0f; });
    const HeightGrid level = MakeGrid([](float, float) { return 0.0f; });
    const HeightGrid crest = MakeGrid([&](float x, float z) { return -0.3f * r2(x, z) / 900.0f; });
    const auto plane = [](float s) { return MakeGrid([s](float x, float) { return s * x; }); };

    float slope = -1.0f;
    CHECK(bowl.Basin(mid, mid, slope) > 0.0f);
    CHECK(slope == Catch::Approx(0.0f).margin(1e-4));
    CHECK(crest.Basin(mid, mid, slope) < 0.0f);
    CHECK(plane(0.05f).Basin(mid, mid, slope) == Catch::Approx(0.0f).margin(1e-4));
    CHECK(slope == Catch::Approx(0.05f).margin(1e-3));

    const float sBowl = WaterShare(bowl, k), sLevel = WaterShare(level, k), sCrest = WaterShare(crest, k);
    const float s3 = WaterShare(plane(0.03f), k), s8 = WaterShare(plane(0.08f), k);
    const float s15 = WaterShare(plane(0.15f), k), s25 = WaterShare(plane(0.25f), k);
    INFO("bowl " << sBowl << " level " << sLevel << " crest " << sCrest << " slopes 3% " << s3 << " 8% " << s8
                 << " 15% " << s15 << " 25% " << s25);
    // Level ground holds puddles; a hollow more, a crest fewer.
    CHECK(sLevel > 0.03f);
    CHECK(sBowl > 1.5f * sLevel);
    CHECK(sCrest < 0.6f * sLevel);
    // The steeper, the fewer; none on a steep slope.
    CHECK(s3 < sLevel);
    CHECK(s8 < 0.6f * s3);
    CHECK(s15 < s8);
    CHECK(s25 == 0.0f);
}

TEST_CASE("Wet weather puddles: their water lines run along no grid line", "[remaster][wet]")
{
    // The micro relief is warped and its octaves turned off the world's axes:
    // its water lines run in every direction. The same octaves on the axes
    // (B7's first noise) drew outlines along the grid's lines.
    const float turned = AxisAlignedShare(PuddleReliefMirror, 0.67f);
    const float axis = AxisAlignedShare(PuddleAxisNoise, 0.67f);
    INFO("turned " << turned << " axis " << axis);
    CHECK(turned < 0.16f);
    CHECK(turned < axis);
    // ... and it stays a field of puddles: the threshold cuts a few out of it.
    int wet = 0, all = 0;
    for (float z = 0.0f; z < 200.0f; z += 0.5f)
        for (float x = 0.0f; x < 200.0f; x += 0.5f, ++all)
            wet += PuddleReliefMirror(x + 3000.0f, z + 5000.0f) > 0.67f ? 1 : 0;
    const float share = static_cast<float>(wet) / all;
    INFO("share " << share);
    CHECK(share > 0.03f);
    CHECK(share < 0.3f);
}

TEST_CASE("Wet weather shaders: only standing water mirrors, the wet ground around it is matte",
          "[remaster][wet][Shaders][ssr]")
{
    const std::string surface = gl::RemasterFragmentPiece(gl::FragmentPiece::Surface);
    const auto constant = [&](const char* name)
    {
        const std::regex re(std::string("const float ") + name + R"( = ([0-9.]+);)");
        std::smatch m;
        INFO(name);
        REQUIRE(std::regex_search(surface, m, re));
        return std::stof(m[1].str());
    };
    // The trace gives the wet weather's ground no reflection outside its
    // puddles (and marches no ray there); B6's own wet ground keeps its own.
    const std::string trace = gl::PostShaderSource(gl::PostShader::SsrTrace);
    CHECK(trace.find("float weight = wetWeather ? puddle : mix(road ? 0.65 * wet : 0.5 * wet * (1.0 - 0.85 * grass), "
                     "1.0, puddle);") != std::string::npos);
    const size_t early = trace.find("if (weight <= 0.0) {");
    REQUIRE(early != std::string::npos);
    CHECK(early < trace.find("int kind = ssrMarch(p, normalize(w2v * rW)"));
    // The forward pass: the ground's film is thin and rough, a broad dim glint
    // near a lamp at most; a puddle keeps its own mirror lobe.
    CHECK(surface.find("if (ground) film *= kWetGroundFilm;") != std::string::npos);
    CHECK(surface.find("float rough = ground ? kWetGroundRough : kWetFilmRough;") != std::string::npos);
    CHECK(constant("kWetGroundFilm") <= 0.4f);
    CHECK(constant("kWetGroundRough") >= 0.45f);
    CHECK(constant("kWetPuddleRough") <= 0.1f);
    CHECK(surface.find("lit += wetLampStreaks(nf, v, kWetPuddleRough) * pud.x;") != std::string::npos);
    // It shows its rain as a darker, deeper colour instead (sealed ground too)
    // and a darker damp rim around each puddle, whose murky water hides the
    // ground's grain.
    CHECK(surface.find("if (ground) soak = mix(max(soak, kWetGroundSeal), soak, veg);") != std::string::npos);
    CHECK(constant("kWetGroundSeal") >= 0.5f);
    CHECK(constant("kWetDampDark") < 0.9f);
    CHECK(surface.find("a = mix(a, toLinear(groundC), pud.x * kWetPuddleMurk);") != std::string::npos);
}
