+float Smoothstep(float a, float b, float x)
+{
+    const float t = Clamp01((x - a) / (b - a));
+    return t * t * (3.0f - 2.0f * t);

+SeaFrameBlend SeaAnimationBlend(float phase, int frames, int bound)
+{
+    SeaFrameBlend b;
+    if (frames <= 1)
+        return b;
+    b.bound = std::max(0, std::min(frames - 1, bound));
+    // Frame k shows alone at phase k / (n - 1); in between, the two around
+    // the phase share it linearly.
+    const float s = Clamp01(phase) * static_cast<float>(frames - 1);
+    const int lo = std::min(static_cast<int>(std::floor(s)), frames - 2);
+    const float f = Clamp01(s - static_cast<float>(lo)); // share of lo + 1
+    if (b.bound == lo)
+    {
+        b.other = lo + 1;
+        b.otherWeight = f;
+    }
+    else if (b.bound == lo + 1)
+    {
+        b.other = lo;
+        b.otherWeight = 1.0f - f;
+    }
+    else
+    {
+        // Neither (AnimateTexture wraps phase 1 exactly to frame 0, and its
+        // rounding floor may bind the frame below at an exact frame edge):
+        // the nearer of the two, through the other unit alone.
+        b.other = f >= 0.5f ? lo + 1 : lo;
+        b.otherWeight = 1.0f;
+    }
+    return b;
+}
+
+float SeaFrameWeight(const SeaFrameBlend& blend, int frame)
+{
+    float w = 0.0f;
+    if (frame == blend.bound)
+        w += 1.0f - blend.otherWeight;
+    if (frame == blend.other)
+        w += blend.otherWeight;
+    return w;
+}
+
+float SeaCrossFadeGain(float w)
+{
+    w = Clamp01(w);
+    const float a = 1.0f - w;
+    const float spread = a * a + w * w + 2.0f * w * a * kSeaFrameCorrelation;
+    return 1.0f / std::sqrt(std::max(spread, 1e-6f));
+}
+
+float SeaHighSun(float sunElevationDeg)
+{
+    return Smoothstep(kSeaHighSunStartDeg, kSeaHighSunFullDeg, sunElevationDeg);
+}
+
+SeaWaveConstants ComputeSeaWaveConstants(float sunElevationDeg, const LightingTuning& tuning,
+                                         const SeaFrameBlend& blend)
+{
+    SeaWaveConstants c;
+    const float high = SeaHighSun(sunElevationDeg);
+    c.relief = std::max(0.0f, tuning.seaRelief);
+    c.crossFade = Clamp01(blend.otherWeight);
+    c.skyMean = kSeaSkyMean * (1.0f + (kSeaSkyMeanNoon - 1.0f) * high) * std::max(0.0f, tuning.seaSky);
+    c.skyTint = kSeaSkyTintNoon * high;
+    c.sheen = (kSeaSheen + (kSeaSheenNoon - kSeaSheen) * high) * std::max(0.0f, tuning.seaSheen);
+    c.sheenFlat = high;
+    return c;
+}

+Rgb SeaSheenLight(const Rgb& rigSun, const Rgb& classicSunDiffuse, float sheen)
+{
+    const float lum = Luminance(rigSun);
+    const float classicLum = Luminance(classicSunDiffuse);
+    const Rgb hue = classicLum > 1e-6f ? classicSunDiffuse * (1.0f / classicLum) : Rgb(1.0f, 1.0f, 1.0f);
+    return hue * (lum * std::max(0.0f, sheen));

+// ---- The sea texture's animation ------------------------------------------

+// The frame PolyProperties::AnimateTexture binds: floor(phase * n), wrapped
+// once (phase 1 exactly wraps to frame 0).
+int AnimateTextureFrame(float phase, int n)
+{
+    int i = static_cast<int>(std::floor(phase * static_cast<float>(n)));
+    if (i < 0)
+        i += n;
+    else if (i >= n)
+        i -= n;
+    return i;
+}

+std::vector<float> Weights(const SeaFrameBlend& b, int n)
+{
+    std::vector<float> w(static_cast<size_t>(n));
+    for (int k = 0; k < n; ++k)
+        w[static_cast<size_t>(k)] = SeaFrameWeight(b, k);
+    return w;
+}

+TEST_CASE("Sea animation: the frames cross-fade continuously instead of switching every 1/n s", "[remaster][SeaWaves]")
+{
+    for (const int n : {2, 7, 8})
+    {
+        INFO("frames " << n);
+        const int steps = 7000;
+        std::vector<float> prev;
+        for (int i = 0; i <= steps; ++i)
+        {
+            const float phase = static_cast<float>(i) / steps;
+            const int bound = AnimateTextureFrame(phase, n);
+            const SeaFrameBlend b = SeaAnimationBlend(phase, n, bound);
+            INFO("phase " << phase << " bound " << bound << " other " << b.other << " w " << b.otherWeight);
+            // The landscape's frame is on unit 0; the backend binds the other.
+            CHECK(b.bound == bound);
+            CHECK(b.other >= 0);
+            CHECK(b.other < n);
+            CHECK(b.otherWeight >= 0.0f);
+            CHECK(b.otherWeight <= 1.0f);
+            const std::vector<float> w = Weights(b, n);
+            float sum = 0.0f;
+            for (float x : w)
+                sum += x;
+            CHECK(sum == Catch::Approx(1.0f).margin(1e-5));
+            // The mix is the even spread: frame k alone at phase k / (n - 1),
+            // the two around the phase in between.
+            const float s = phase * static_cast<float>(n - 1);
+            for (int k = 0; k < n; ++k)
+            {
+                const float want = std::max(0.0f, 1.0f - std::fabs(s - static_cast<float>(k)));
+                CHECK(w[static_cast<size_t>(k)] == Catch::Approx(want).margin(1e-4));
+            }
+            // Continuous: no frame's share jumps between neighbouring phases
+            // (a hard switch moves a whole frame's share at once).
+            if (!prev.empty())
+            {
+                const float step = static_cast<float>(n - 1) / steps;
+                for (int k = 0; k < n; ++k)
+                    CHECK(std::fabs(w[static_cast<size_t>(k)] - prev[static_cast<size_t>(k)]) <= step * 1.01f + 1e-5f);
+            }
+            prev = w;
+        }
+    }
